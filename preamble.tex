% Page limit per part: \checkpartlength{<Name>} at the end of a part compares
% its length (from \label{part:<Name>}) against the limit, and flags it in the
% log and in red in the PDF when exceeded.
\usepackage{refcount}
\newcommand{\partpagelimit}{3}
\newcommand{\checkpartlength}[1]{%
    \label{partend:#1}%
    \ifnum\numexpr\getpagerefnumber{partend:#1}-\getpagerefnumber{part:#1}+1\relax>\partpagelimit\relax
        \PackageWarningNoLine{tcx3901}{Part '#1' exceeds the
            \partpagelimit-page limit}%
        \par\textcolor{red}{\bfseries This part exceeds the
            \partpagelimit-page limit.}%
    \fi}
